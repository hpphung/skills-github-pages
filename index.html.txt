  <script>
    const RSS_PROXY = "https://api.rss2json.com/v1/api.json?rss_url=";

    const feeds = {
      crypto: [
        "https://cointelegraph.com/rss",
        "https://www.coindesk.com/arc/outboundfeeds/rss/"
      ],
      vnStock: [
        "https://cafef.vn/thi-truong-chung-khoan.rss",
        "https://vneconomy.vn/chung-khoan.rss"
      ],
      globalStock: [
        "https://www.investing.com/rss/news_25.rss",
        "https://www.marketwatch.com/rss/topstories"
      ]
    };

    async function loadFeed(feedUrl, containerId) {
      try {
        const res = await fetch(RSS_PROXY + encodeURIComponent(feedUrl));
        const data = await res.json();
        const container = document.getElementById(containerId);

        data.items.slice(0, 5).forEach(item => {
          const li = document.createElement("li");
          li.innerHTML = `
            <a href="${item.link}" target="_blank" class="block bg-white p-3 rounded-lg shadow hover:bg-gray-100">
              <div class="font-medium">${item.title}</div>
              <div class="text-xs text-gray-500">${data.feed.title}</div>
            </a>`;
          container.appendChild(li);
        });
      } catch (e) {
        console.error("RSS error", e);
      }
    }

    feeds.crypto.forEach(url => loadFeed(url, "cryptoNews"));
    feeds.vnStock.forEach(url => loadFeed(url, "vnStockNews"));
    feeds.globalStock.forEach(url => loadFeed(url, "globalStockNews"));
  </script>
